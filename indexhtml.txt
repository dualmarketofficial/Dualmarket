  <script>
    function openModal(id) {
      const modal = document.getElementById(id);
      if (modal) modal.style.display = 'flex';
    }

    function closeModal(id) {
      const modal = document.getElementById(id);
      if (modal) modal.style.display = 'none';
    }

    function setActiveTab(tab) {
      document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));
      if(tab === 'home') {
        document.getElementById('nav-home').classList.add('active');
        loadListings();
      } else if(tab === 'updates') {
        document.getElementById('nav-updates').classList.add('active');
        loadBoostedListings();
      } else if(tab === 'messages') {
        document.getElementById('nav-messages').classList.add('active');
        alert("Direct WhatsApp community inquiries active.");
      }
    }

    const SUPABASE_URL = "https://byufsxfibjiczvmjcccm.supabase.co";
    const SUPABASE_KEY = "sb_publishable_SY5YfiZKo8v2__CIrlHhJQ_3It2Rx0C"; 
    
    let supabase = null;
    try {
      if (window.supabase && typeof window.supabase.createClient === 'function') {
        supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY);
      }
    } catch (e) {
      console.error("Supabase init error:", e);
    }

    async function loadListings() {
      const grid = document.getElementById('listingsGrid');
      if (!supabase) {
        grid.innerHTML = '<p style="color:var(--text-muted); text-align:center;">No listings available yet.</p>';
        return;
      }

      try {
        const { data, error } = await supabase
          .from('listings')
          .select('*')
          .eq('is_approved', true)
          .order('is_boosted', { ascending: false })
          .order('created_at', { ascending: false });

        if (error || !data || data.length === 0) {
          grid.innerHTML = '<p style="color:var(--text-muted); text-align:center;">No active listings yet. Be the first to post using the Add button below!</p>';
          return;
        }

        renderGrid(data);
      } catch (err) {
        grid.innerHTML = '<p style="color:var(--text-muted); text-align:center;">Failed to load listings.</p>';
      }
    }

    async function loadBoostedListings() {
      const grid = document.getElementById('listingsGrid');
      if (!supabase) return;

      const { data } = await supabase
        .from('listings')
        .select('*')
        .eq('is_approved', true)
        .eq('is_boosted', true);

      if (!data || data.length === 0) {
        grid.innerHTML = '<p style="color:var(--text-muted); text-align:center;">🚀 No active rocket updates or pinned listings right now.</p>';
        return;
      }
      renderGrid(data);
    }

    function renderGrid(data) {
      const grid = document.getElementById('listingsGrid');
      grid.innerHTML = data.map(item => {
        const cleanPhone = item.whatsapp_number ? item.whatsapp_number.replace(/[^0-9]/g, '') : '';
        const waLink = `https://wa.me/${cleanPhone.startsWith('263') ? cleanPhone : '263' + cleanPhone.replace(/^0/, '')}?text=Hi%20${encodeURIComponent(item.title)},%20I%20saw%20your%20listing%20on%20DualMarket.`;
        
        return `
          <div class="card ${item.is_boosted ? 'boosted-card' : ''}">
            <div class="card-img-wrap">
              <span class="category-tag">${item.category}</span>
              ${item.is_boosted ? '<span class="boosted-badge">🚀 ROCKET DROP</span>' : ''}
              <img src="https://images.unsplash.com/photo-1555396273-367ea4eb4db5?auto=format&fit=crop&w=600&q=80" class="card-img" alt="${item.title}">
            </div>
            <div class="card-body">
              <div class="location-rating">
                <span>📍 ${item.location}</span>
                <span style="color:var(--orange);">✔ Verified</span>
              </div>
              <h3 class="card-title">${item.title}</h3>
              <p class="card-desc">${item.description}</p>
              <div class="card-footer">
                <div>
                  <div style="font-size:9px; color:var(--text-muted); font-weight:700;">PRICE</div>
                  <div class="price-val">${item.price_info || 'Inquire'}</div>
                </div>
                <a href="${waLink}" target="_blank" class="btn-whatsapp">WhatsApp</a>
              </div>
            </div>
          </div>
        `;
      }).join('');
    }

    document.getElementById('vendorForm').addEventListener('submit', async function(e) {
      e.preventDefault();
      if (!supabase) return alert("Database connection error.");

      const user_email = document.getElementById('vendorEmail').value;
      const title = document.getElementById('postTitle').value;
      const category = document.getElementById('postCategory').value;
      const location = document.getElementById('postLocation').value;
      const whatsapp_number = document.getElementById('postWhatsapp').value;
      const price_info = document.getElementById('postPrice').value;
      const description = document.getElementById('postDesc').value;

      const { error } = await supabase.from('listings').insert([
        { user_email, title, category, location, whatsapp_number, price_info, description, is_approved: false, is_boosted: false }
      ]);

      if (error) {
        alert("Error: " + error.message);
      } else {
        alert("🎉 Success! Submitted for Admin approval.");
        closeModal('postModal');
        this.reset();
      }
    });

    function checkAdminAuth() {
      const code = document.getElementById('adminPasscode').value;
      if (code === '263788') {
        document.getElementById('adminAuthBlock').style.display = 'none';
        document.getElementById('adminContent').style.display = 'block';
        loadAdminQueue();
      } else {
        alert("Incorrect Admin Code!");
      }
    }

    async function loadAdminQueue() {
      const queue = document.getElementById('adminListingsQueue');
      const { data, error } = await supabase.from('listings').select('*').order('created_at', { ascending: false });

      if (error || !data || data.length === 0) {
        queue.innerHTML = '<p style="color:var(--text-muted);">No submissions found.</p>';
        return;
      }

      queue.innerHTML = data.map(item => `
        <div class="admin-item">
          <strong>${item.title}</strong> (${item.category} - ${item.location})<br>
          <small>Email: ${item.user_email} | WA: ${item.whatsapp_number}</small><br>
          <small>Status: ${item.is_approved ? '✅ Approved' : '⏳ Pending'} | Boosted: ${item.is_boosted ? '🚀 Yes' : 'No'}</small>
          <div class="admin-actions">
            ${!item.is_approved ? `<button onclick="toggleApprove('${item.id}', true)" class="btn-approve">Approve</button>` : `<button onclick="toggleApprove('${item.id}', false)" class="btn-block">Unapprove</button>`}
            <button onclick="toggleBoost('${item.id}', ${!item.is_boosted})" class="btn-pin">${item.is_boosted ? 'Unpin' : 'Rocket Boost'}</button>
            <button onclick="deleteListing('${item.id}')" class="btn-block">Delete</button>
          </div>
        </div>
      `).join('');
    }

    async function toggleApprove(id, status) {
      await supabase.from('listings').update({ is_approved: status }).eq('id', id);
      loadAdminQueue();
      loadListings();
    }

    async function toggleBoost(id, status) {
      await supabase.from('listings').update({ is_boosted: status }).eq('id', id);
      loadAdminQueue();
      loadListings();
    }

    async function deleteListing(id) {
      if (confirm("Delete this listing permanently?")) {
        await supabase.from('listings').delete().eq('id', id);
        loadAdminQueue();
        loadListings();
      }
    }

    loadListings();
  </script>
